SERP01.sparql
SERP01: ¿Cuál es el identificador de cada servicio público?
PREFIX serpub: <https://edint.es/def/servicios-publicos#>

SELECT ?servicioPublico ?identificador
WHERE {
  ?servicioPublico a serpub:PublicService ;
                   serpub:identifier ?identificador .
}
ORDER BY ?servicioPublico
SERP02.sparql
SERP02: ¿A qué tipo pertenece cada servicio público?
PREFIX serpub: <https://edint.es/def/servicios-publicos#>
PREFIX skos: <http://www.w3.org/2004/02/skos/core#>

SELECT ?servicioPublico ?tipoServicio ?etiquetaTipoServicio
WHERE {
  ?servicioPublico a serpub:PublicService ;
                   serpub:typePublicService ?tipoServicio .

  ?tipoServicio skos:inScheme <http://vocab.linkeddata.es/datosabiertos/kos/edint/PublicServices> .

  OPTIONAL {
    ?tipoServicio skos:prefLabel ?etiquetaTipoServicio .
    FILTER(LANG(?etiquetaTipoServicio) = "es")
  }
}
ORDER BY ?servicioPublico ?etiquetaTipoServicio
SERP03.sparql
SERP03: ¿Qué procedimientos administrativos están asociados a cada servicio público?
PREFIX serpub: <https://edint.es/def/servicios-publicos#>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT ?servicioPublico ?etiquetaServicio ?procedimiento ?etiquetaProcedimiento
WHERE {
  ?servicioPublico a serpub:PublicService ;
                   serpub:hasAdministrativeProcedure ?procedimiento .

  ?procedimiento a serpub:AdministrativeProcedure .

  OPTIONAL {
    ?servicioPublico rdfs:label ?etiquetaServicio .
    FILTER(LANG(?etiquetaServicio) = "es")
  }

  OPTIONAL {
    ?procedimiento rdfs:label ?etiquetaProcedimiento .
    FILTER(LANG(?etiquetaProcedimiento) = "es")
  }
}
ORDER BY ?etiquetaServicio ?etiquetaProcedimiento

SERP06.sparql
SERP06: ¿Qué organización pública es la autoridad competente de cada servicio público?
PREFIX serpub: <https://edint.es/def/servicios-publicos#>
PREFIX m8g: <http://data.europa.eu/m8g/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT ?servicioPublico ?etiquetaServicio ?organizacionPublica ?etiquetaOrganizacion
WHERE {
  ?servicioPublico a serpub:PublicService ;
                   serpub:hasCompetentAuthority ?organizacionPublica .

  ?organizacionPublica a m8g:PublicOrganisation .

  OPTIONAL {
    ?servicioPublico rdfs:label ?etiquetaServicio .
    FILTER(LANG(?etiquetaServicio) = "es")
  }

  OPTIONAL {
    ?organizacionPublica rdfs:label ?etiquetaOrganizacion .
    FILTER(LANG(?etiquetaOrganizacion) = "es")
  }
}
ORDER BY ?etiquetaOrganizacion ?etiquetaServicio

SERP09.sparql
SERP09: ¿Qué reglas sigue cada servicio público?
PREFIX serpub: <https://edint.es/def/servicios-publicos#>
PREFIX cpsv: <http://purl.org/vocab/cpsv#>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT ?servicioPublico ?etiquetaServicio ?regla ?etiquetaRegla
WHERE {
  ?servicioPublico a serpub:PublicService ;
                   cpsv:follows ?regla .

  ?regla a cpsv:Rule .

  OPTIONAL {
    ?servicioPublico rdfs:label ?etiquetaServicio .
    FILTER(LANG(?etiquetaServicio) = "es")
  }

  OPTIONAL {
    ?regla rdfs:label ?etiquetaRegla .
    FILTER(LANG(?etiquetaRegla) = "es")
  }
}
ORDER BY ?etiquetaServicio ?etiquetaRegla
